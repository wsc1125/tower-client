# 高塔爬塔玩法实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把「主菜单 + 单房间战斗沙盒」改造成完整的爬塔循环：出发 → 逐层推进（敌人随层数变强、掉落随层数变好）→ 撤退层二选一（继续爬 / 撤退结算）→ 回主菜单；死亡清空全部库存。

**Architecture:** 沿用既有「纯逻辑类 + MonoBehaviour 接线」分层。`Gameplay` 放塔流程与生成逻辑；`BattleHUD` 放结算面板 UI。结算面板通过纯展示数据 `TowerResultEntry` + `Confirmed` 事件与 `Gameplay` 解耦，因为依赖方向是 `Gameplay → BattleHUD → Inventory`（不可逆，`BattleHUD` 不能反向引用 `Gameplay`）。难度与掉落全部由 `TowerConfig`（ScriptableObject）驱动，`FloorGenerator` 做成纯静态函数以便 EditMode 直接验证。

**Tech Stack:** Unity 2022.3.62f3c1、C#、Unity Test Framework（EditMode）、TextMeshPro、Unity MCP（编辑器接线与测试执行，端口 7890）

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;

/// <summary>
/// 爬塔的难度与掉落配置：所有数值都在这里调，改曲线不用重编译。
/// 层数从 1 开始；难度按「每层线性递增」计算，掉落按层数区间分档。
/// </summary>
[CreateAssetMenu(menuName = "Gameplay/Tower Config", fileName = "TowerConfig")]
public sealed class TowerConfig : ScriptableObject
{
    [Serializable]
    public struct EnemyWeight
    {
        public GameObject prefab;
        public int weight;
    }

    [Serializable]
    public struct LootTier
    {
        [Tooltip("该档位的起始层数（含）。")]
        public int fromFloor;
        public LootTable table;
    }

    [Header("难度曲线")]
    [Tooltip("每高一层，敌人血量增加的比例。0.15 表示第 10 层为 1 + 0.15 × 9 = 2.35 倍。")]
    [SerializeField] private float healthGrowthPerFloor = 0.15f;

    [Tooltip("每高一层，敌人伤害增加的比例。")]
    [SerializeField] private float damageGrowthPerFloor = 0.10f;

    [Tooltip("每多少层多刷一只敌人。")]
    [SerializeField] private int floorsPerExtraEnemy = 2;

    [Tooltip("单层敌人数量上限。")]
    [SerializeField] private int maxEnemiesPerFloor = 6;

    [Header("撤退层")]
    [Tooltip("每隔多少层出现一个撤退层。5 表示第 5、10、15… 层。")]
    [SerializeField] private int retreatEveryFloors = 5;

    [Header("敌人池")]
    [Tooltip("按权重随机抽取。prefab 为空或 weight <= 0 的条目会被跳过。")]
    [SerializeField] private EnemyWeight[] enemyPool;

    [Header("掉落分档")]
    [Tooltip("按 fromFloor 升序填写。取 fromFloor <= 当前层数 中最大的那一档。")]
    [SerializeField] private LootTier[] lootTiers;

    public float HealthGrowthPerFloor => healthGrowthPerFloor;
    public float DamageGrowthPerFloor => damageGrowthPerFloor;
    public int FloorsPerExtraEnemy => floorsPerExtraEnemy > 0 ? floorsPerExtraEnemy : 1;
    public int MaxEnemiesPerFloor => maxEnemiesPerFloor > 0 ? maxEnemiesPerFloor : 1;
    public int RetreatEveryFloors => retreatEveryFloors;

    public IReadOnlyList<EnemyWeight> EnemyPool => enemyPool ?? Array.Empty<EnemyWeight>();
    public IReadOnlyList<LootTier> LootTiers => lootTiers ?? Array.Empty<LootTier>();

    /// <summary>测试与运行时接线用；不改动 Inspector 里已序列化的数据以外的行为。</summary>
    public void Configure(EnemyWeight[] pool, LootTier[] tiers)
    {
        enemyPool = pool;
        lootTiers = tiers;
    }
}
```

```csharp
using System.Collections.Generic;
using UnityEngine;

/// <summary>
/// 由层数推导该层内容：敌人数量与种类、难度系数、掉落表、是否撤退层。
/// 全部是静态纯函数，接收 <see cref="System.Random"/> 而非 UnityEngine.Random，
/// 因此 EditMode 可用固定种子验证可复现性，也为将来把权威随机移到服务器留下唯一改动点。
/// </summary>
public static class FloorGenerator
{
    /// <summary>生成一层的完整描述。config 为 null 时返回空层（不抛异常）。</summary>
    public static FloorPlan Generate(int floor, TowerConfig config, System.Random random)
    {
        var resolvedFloor = Mathf.Max(1, floor);
        if (config == null)
            return new FloorPlan(resolvedFloor, false, System.Array.Empty<GameObject>(), null);

        var count = ResolveEnemyCount(resolvedFloor, config.FloorsPerExtraEnemy, config.MaxEnemiesPerFloor);
        var enemies = new GameObject[count];
        for (var i = 0; i < count; i++)
            enemies[i] = PickEnemy(config.EnemyPool, random);

        return new FloorPlan(
            resolvedFloor,
            IsRetreatFloor(resolvedFloor, config.RetreatEveryFloors),
            enemies,
            ResolveLootTable(resolvedFloor, config.LootTiers));
    }

    /// <summary>敌人数量：第 1 层 1 只，每 floorsPerExtraEnemy 层多 1 只，上限 max。</summary>
    public static int ResolveEnemyCount(int floor, int floorsPerExtraEnemy, int max)
    {
        var resolvedFloor = Mathf.Max(1, floor);
        var step = Mathf.Max(1, floorsPerExtraEnemy);
        var count = 1 + (resolvedFloor - 1) / step;
        return Mathf.Clamp(count, 1, Mathf.Max(1, max));
    }

    /// <summary>血量系数：1 + growth × (层数 - 1)。第 1 层恒为 1。</summary>
    public static float ResolveHealthMultiplier(int floor, float growthPerFloor)
    {
        return 1f + growthPerFloor * (Mathf.Max(1, floor) - 1);
    }

    /// <summary>伤害系数：1 + growth × (层数 - 1)。第 1 层恒为 1。</summary>
    public static float ResolveDamageMultiplier(int floor, float growthPerFloor)
    {
        return 1f + growthPerFloor * (Mathf.Max(1, floor) - 1);
    }

    /// <summary>是否是撤退层。间隔非正数时恒为 false。</summary>
    public static bool IsRetreatFloor(int floor, int everyFloors)
    {
        if (everyFloors <= 0)
            return false;

        return Mathf.Max(1, floor) % everyFloors == 0;
    }

    /// <summary>取 fromFloor &lt;= 层数 中 fromFloor 最大的那一档；没有匹配档时返回 null。</summary>
    public static LootTable ResolveLootTable(int floor, IReadOnlyList<TowerConfig.LootTier> tiers)
    {
        if (tiers == null)
            return null;

        var resolvedFloor = Mathf.Max(1, floor);
        LootTable best = null;
        var bestFrom = int.MinValue;

        for (var i = 0; i < tiers.Count; i++)
        {
            var tier = tiers[i];
            if (tier.table == null || tier.fromFloor > resolvedFloor)
                continue;

            if (tier.fromFloor <= bestFrom)
                continue;

            bestFrom = tier.fromFloor;
            best = tier.table;
        }

        return best;
    }

    /// <summary>按权重加权随机抽一个敌人。池为空、总权重为 0 或 random 为 null 时返回 null。</summary>
    public static GameObject PickEnemy(IReadOnlyList<TowerConfig.EnemyWeight> pool, System.Random random)
    {
        if (pool == null || random == null)
            return null;

        var total = 0;
        for (var i = 0; i < pool.Count; i++)
        {
            var entry = pool[i];
            if (entry.prefab == null || entry.weight <= 0)
                continue;

            total += entry.weight;
        }

        if (total <= 0)
            return null;

        var roll = random.Next(total);
        for (var i = 0; i < pool.Count; i++)
        {
            var entry = pool[i];
            if (entry.prefab == null || entry.weight <= 0)
                continue;

            if (roll < entry.weight)
                return entry.prefab;

            roll -= entry.weight;
        }

        return null;
    }
}
```

```csharp
using System;
using System.Collections.Generic;

/// <summary>
/// 一局爬塔的局内状态：当前层数、本局拾取的战利品清单、是否已结束。
/// 纯 C# 类，不继承 MonoBehaviour，便于在 EditMode 直接验证。
/// 只活在本次战斗场景内，退出即丢（本轮不做持久化）。
/// </summary>
public sealed class TowerRunState
{
    private readonly List<ItemDefinition> collectedLoot = new List<ItemDefinition>();

    /// <summary>当前层数，从 1 开始。</summary>
    public int CurrentFloor { get; private set; } = 1;

    /// <summary>本局是否已结束（撤退或死亡）。结束后传送点与死亡回调都不再动作。</summary>
    public bool IsFinished { get; private set; }

    /// <summary>本局拾取的全部战利品，按拾取顺序。实际物品在 Inventory.carry 里，这里只用于结算展示。</summary>
    public IReadOnlyList<ItemDefinition> CollectedLoot => collectedLoot;

    /// <summary>开始新的一局：层数归 1，清空战利品与结束标记。</summary>
    public void BeginRun()
    {
        CurrentFloor = 1;
        IsFinished = false;
        collectedLoot.Clear();
    }

    /// <summary>进入下一层。</summary>
    public void AdvanceFloor()
    {
        CurrentFloor++;
    }

    /// <summary>记一件本局战利品。null 被忽略。</summary>
    public void RecordLoot(ItemDefinition item)
    {
        if (item == null)
            return;

        collectedLoot.Add(item);
    }

    /// <summary>清空本局战利品清单（死亡时调用）。</summary>
    public void ClearLoot()
    {
        collectedLoot.Clear();
    }

    /// <summary>标记本局结束。幂等。</summary>
    public void Finish()
    {
        IsFinished = true;
    }
}
```

```csharp
using System;
using UnityEngine;

/// <summary>传送点的种类。</summary>
public enum PortalKind
{
    /// <summary>进入下一层。</summary>
    Next,

    /// <summary>撤退结算。</summary>
    Retreat,
}

/// <summary>
/// 层间传送点：玩家踩上去触发一次，把「踩到了哪种传送点」抛给订阅者。
/// 只抛事件，不认识 <see cref="TowerRunManager"/>，由 Manager 在生成时订阅。
/// </summary>
[RequireComponent(typeof(Collider2D))]
public sealed class FloorPortal2D : MonoBehaviour
{
    [SerializeField] private PortalKind kind;
    [SerializeField] private SpriteRenderer visual;

    private Collider2D trigger;
    private bool consumed;

    public PortalKind Kind => kind;

    /// <summary>玩家踩上来时触发一次，参数为传送点种类。</summary>
    public event Action<PortalKind> Entered;

    /// <summary>测试与运行时接线用。</summary>
    public void Configure(PortalKind value)
    {
        kind = value;
    }

    private void Awake()
    {
        trigger = GetComponent<Collider2D>();
        if (trigger != null)
            trigger.isTrigger = true;
    }

    private void OnTriggerEnter2D(Collider2D other)
    {
        TryEnter(other);
    }

    /// <summary>
    /// 判定并触发。拆成公开方法以便 EditMode 直接调用（EditMode 下 Physics2D 查不到新建的碰撞体）。
    /// 只认玩家；已触发过的传送点不再响应。
    /// </summary>
    public bool TryEnter(Collider2D other)
    {
        if (consumed || other == null)
            return false;

        if (other.GetComponentInParent<PlayerMovement2D>() == null)
            return false;

        consumed = true;

        // 立刻关掉碰撞体，避免同一帧被多个碰撞体重复触发。
        if (trigger != null)
            trigger.enabled = false;

        Entered?.Invoke(kind);
        return true;
    }
}
```

新建 `Assets/Gameplay/Runtime/TowerRunManager.cs`：

```csharp
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.SceneManagement;

/// <summary>
/// 爬塔流程编排：生成一层 → 统计清空 → 生成传送点 → 换层 → 撤退 / 死亡收尾。
/// 换层时只销毁房间与敌人，玩家对象跨层保留，因此血量与护甲天然继承到下一层
/// （这是难度曲线生效的前提）。
/// </summary>
public sealed class TowerRunManager : MonoBehaviour
{
    [Header("配置")]
    [SerializeField] private TowerConfig config;
    [SerializeField] private GameObject roomPrefab;
    [SerializeField] private Transform roomParent;
    [SerializeField] private FloorPortal2D nextPortalPrefab;
    [SerializeField] private FloorPortal2D retreatPortalPrefab;

    [Header("接线")]
    [SerializeField] private Transform player;
    [SerializeField] private CameraFollow2D cameraFollow;
    [SerializeField] private StatusPanelController statusPanel;
    [SerializeField] private TowerResultPanelController resultPanel;
    [SerializeField] private InventoryHolder inventoryHolder;
    [SerializeField] private PlayerHealth2D playerHealth;
    [SerializeField] private string mainMenuSceneName = "MainMenu";

    [Header("节奏")]
    [SerializeField] private float portalSpawnDelay = 0.6f;
    [SerializeField] private float clearCheckInterval = 0.2f;
    [SerializeField] private float enemySpawnRadiusMin = 3.5f;
    [SerializeField] private float enemySpawnRadiusMax = 5.5f;
    [SerializeField] private float portalOffsetX = 2.5f;

    private readonly TowerRunState state = new TowerRunState();
    private readonly List<GameObject> spawnedEnemies = new List<GameObject>();
    private readonly List<FloorPortal2D> spawnedPortals = new List<FloorPortal2D>();
    private GameObject currentRoom;
    private System.Random random;
    private float clearTimer;
    private bool portalsSpawned;
    private bool subscribedDeath;

    public TowerRunState State => state;

    private void Awake()
    {
        random = new System.Random();
    }

    private void OnEnable()
    {
        if (playerHealth != null)
        {
            playerHealth.Died += HandlePlayerDied;
            subscribedDeath = true;
        }
    }

    private void OnDisable()
    {
        if (subscribedDeath && playerHealth != null)
        {
            playerHealth.Died -= HandlePlayerDied;
            subscribedDeath = false;
        }
    }

    private void Start()
    {
        state.BeginRun();
        BuildFloor(state.CurrentFloor);
    }

    private void Update()
    {
        if (state.IsFinished || portalsSpawned)
            return;

        clearTimer -= Time.deltaTime;
        if (clearTimer > 0f)
            return;

        clearTimer = Mathf.Max(0.05f, clearCheckInterval);
        if (IsFloorCleared())
            StartCoroutine(SpawnPortalsAfterDelay());
    }

    /// <summary>生成指定层：清掉上一层残留 → 实例化房间 → 放玩家 → 生成敌人 → 刷新 HUD。</summary>
    public void BuildFloor(int floor)
    {
        ClearFloor();

        if (roomPrefab != null)
        {
            var parent = roomParent != null ? roomParent : transform;
            currentRoom = Instantiate(roomPrefab, Vector3.zero, Quaternion.identity, parent);
        }

        var center = currentRoom != null ? (Vector2)currentRoom.transform.position : Vector2.zero;
        MovePlayerTo(center);

        if (cameraFollow != null)
            cameraFollow.CaptureRoomBounds();

        var plan = FloorGenerator.Generate(floor, config, random);
        SpawnEnemies(plan, center);

        if (statusPanel != null)
            statusPanel.SetLevel(floor);
    }

    /// <summary>本层敌人是否已全部清空。虫群（母巢召唤）也是 IDamageable，必须一并计入。</summary>
    public bool IsFloorCleared()
    {
        foreach (var enemy in spawnedEnemies)
        {
            if (enemy == null)
                continue;

            var damageable = enemy.GetComponent<IDamageable>();
            if (damageable != null && !damageable.IsDead)
                return false;
        }

        // 母巢召唤的虫群不在 spawnedEnemies 里，单独扫一遍。
        foreach (var swarm in FindObjectsOfType<SwarmChaser2D>())
        {
            if (swarm != null && !swarm.IsDead)
                return false;
        }

        return true;
    }

    /// <summary>玩家踩到传送点。</summary>
    public void HandlePortalEntered(PortalKind kind)
    {
        if (state.IsFinished)
            return;

        if (kind == PortalKind.Retreat)
        {
            Retreat();
            return;
        }

        state.AdvanceFloor();
        BuildFloor(state.CurrentFloor);
    }

    /// <summary>撤退：结束本局并弹出结算面板。</summary>
    public void Retreat()
    {
        if (state.IsFinished)
            return;

        state.Finish();
        ClearFloor();

        if (resultPanel != null)
            resultPanel.Show(BuildResultEntries(), TotalLootValue(), state.CurrentFloor);
    }

    /// <summary>本局中止（玩家死亡）。只标记结束并清空战利品清单，切场景由 PlayerDeathFlow2D 负责。</summary>
    public void AbortRun()
    {
        state.Finish();
        state.ClearLoot();
    }

    /// <summary>把本局战利品清单转成结算面板的展示数据。</summary>
    public IReadOnlyList<TowerResultEntry> BuildResultEntries()
    {
        var entries = new List<TowerResultEntry>();
        foreach (var item in state.CollectedLoot)
        {
            if (item == null)
                continue;

            entries.Add(new TowerResultEntry
            {
                DisplayName = item.DisplayName,
                Value = item.Value,
                Icon = item.Icon,
            });
        }

        return entries;
    }

    /// <summary>本局战利品的总价值。</summary>
    public int TotalLootValue()
    {
        var total = 0;
        foreach (var item in state.CollectedLoot)
        {
            if (item != null)
                total += item.Value;
        }

        return total;
    }

    /// <summary>结算面板确认后回主菜单。</summary>
    public void ConfirmRetreat()
    {
        SceneManager.LoadScene(mainMenuSceneName);
    }

    private void HandlePlayerDied()
    {
        AbortRun();
    }

    private IEnumerator SpawnPortalsAfterDelay()
    {
        portalsSpawned = true;
        yield return new WaitForSeconds(Mathf.Max(0f, portalSpawnDelay));

        if (state.IsFinished)
            yield break;

        var center = currentRoom != null ? (Vector2)currentRoom.transform.position : Vector2.zero;
        var isRetreatFloor = FloorGenerator.IsRetreatFloor(state.CurrentFloor, config != null ? config.RetreatEveryFloors : 0);

        if (isRetreatFloor && retreatPortalPrefab != null)
        {
            SpawnPortal(retreatPortalPrefab, center + new Vector2(-portalOffsetX, 0f));
            SpawnPortal(nextPortalPrefab, center + new Vector2(portalOffsetX, 0f));
            yield break;
        }

        SpawnPortal(nextPortalPrefab, center);
    }

    private void SpawnPortal(FloorPortal2D prefab, Vector2 position)
    {
        if (prefab == null)
            return;

        var portal = Instantiate(prefab, position, Quaternion.identity);
        portal.Entered += HandlePortalEntered;
        spawnedPortals.Add(portal);
    }

    private void SpawnEnemies(FloorPlan plan, Vector2 center)
    {
        for (var i = 0; i < plan.Enemies.Length; i++)
        {
            var prefab = plan.Enemies[i];
            if (prefab == null)
                continue;

            var angle = 360f * i / Mathf.Max(1, plan.Enemies.Length);
            var radius = Random.Range(enemySpawnRadiusMin, enemySpawnRadiusMax);
            var offset = Quaternion.Euler(0f, 0f, angle) * (Vector3)(Vector2.right * radius);
            var position = center + (Vector2)offset;

            var enemy = Instantiate(prefab, position, Quaternion.identity);

            // 先 SetActive(false) 再缩放再 SetActive(true)：Awake 在 SetActive(true) 时才跑，
            // 因此缩放先生效，EnemyBrain2D 会用缩放后的血量创建 health。
            enemy.SetActive(false);
            var scaler = enemy.GetComponent<EnemyStatScaler>();
            if (scaler == null)
                scaler = enemy.AddComponent<EnemyStatScaler>();
            scaler.Apply(plan.Floor, config);
            enemy.SetActive(true);

            spawnedEnemies.Add(enemy);
        }
    }

    private void MovePlayerTo(Vector2 position)
    {
        if (player == null)
            return;

        var body = player.GetComponent<Rigidbody2D>();
        if (body != null)
            body.position = position;

        player.position = new Vector3(position.x, position.y, player.position.z);
    }

    private void ClearFloor()
    {
        foreach (var portal in spawnedPortals)
        {
            if (portal != null)
            {
                portal.Entered -= HandlePortalEntered;
                Destroy(portal.gameObject);
            }
        }

        spawnedPortals.Clear();

        foreach (var enemy in spawnedEnemies)
        {
            if (enemy != null)
                Destroy(enemy);
        }

        spawnedEnemies.Clear();

        // 母巢召唤的虫群不在 spawnedEnemies 里，换层时一并清掉。
        foreach (var swarm in FindObjectsOfType<SwarmChaser2D>())
        {
            if (swarm != null)
                Destroy(swarm.gameObject);
        }

        if (currentRoom != null)
        {
            Destroy(currentRoom);
            currentRoom = null;
        }

        portalsSpawned = false;
        clearTimer = 0f;
    }
}
```

```csharp
    /// <summary>
    /// 收集场上所有可受伤目标。必须覆盖全部敌人类型（含虫群与 Boss），
    /// 否则「附近有敌人」判定会漏，战斗中点开拾取列表。
    /// 与 RifleShooter2D.FindNearestEnemy 同构，避免 FindObjectsOfType&lt;MonoBehaviour&gt; 的全量扫描。
    /// </summary>
    private static IDamageable[] CollectDamageables()
    {
        var brains = FindObjectsOfType<EnemyBrain2D>(false);
        var swarms = FindObjectsOfType<SwarmChaser2D>(false);
        var bosses = FindObjectsOfType<BossBrain2D>(false);

        var result = new IDamageable[brains.Length + swarms.Length + bosses.Length];
        var index = 0;
        for (var i = 0; i < brains.Length; i++)
            result[index++] = brains[i];
        for (var i = 0; i < swarms.Length; i++)
            result[index++] = swarms[i];
        for (var i = 0; i < bosses.Length; i++)
            result[index++] = bosses[i];

        return result;
    }
```
