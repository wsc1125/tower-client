# 掉落与拾取系统实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 敌人死亡后按掉落表掷点生成地面掉落物；玩家在附近无存活敌人时，HUD 弹出「附近物品」列表，玩家从列表中选择拾取，物品进入现有背包。

**Architecture:** 沿用既有「纯逻辑类 + MonoBehaviour 接线」分层。`Inventory` 程序集放物品数据（`ItemDefinition`、`LootTable`）；`Gameplay` 放掉落与交互逻辑；`BattleHUD` 放列表 UI。UI 通过纯展示数据 `LootEntry` + `ItemSelected` 事件与 `Gameplay` 解耦，因为依赖方向是 `Gameplay → BattleHUD → Inventory`（不可逆，`BattleHUD` 不能反向引用 `Gameplay`）。

**Tech Stack:** Unity 2022.3.62f3c1、C#、Unity Test Framework（EditMode）、TextMeshPro、Unity MCP（编辑器接线与测试执行，端口 7890）

## Global Constraints

- 不改动 `Inventory` 的 20 格容量、装备槽与拖拽逻辑

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;

[CreateAssetMenu(menuName = "Inventory/Loot Table", fileName = "LootTable")]
public sealed class LootTable : ScriptableObject
{
    [Serializable]
    public struct Entry
    {
        public ItemDefinition item;
        [Range(0f, 1f)] public float dropChance;
    }

    [SerializeField] private Entry[] entries;

    public Entry[] Entries => entries;

    /// 测试与运行时接线用；不影响 Inspector 中已有的序列化数据。
    public void Configure(Entry[] source)
    {
        entries = source;
    }

    /// 每条独立掷点，互不影响，因此一次死亡可能命中多件。
    public IReadOnlyList<ItemDefinition> Roll(System.Random random)
    {
        var result = new List<ItemDefinition>();
        if (entries == null || random == null)
            return result;

        foreach (var entry in entries)
        {
            if (entry.item == null)
                continue;
            if (!ShouldDrop(entry.dropChance, random.NextDouble()))
                continue;

            result.Add(entry.item);
        }

        return result;
    }

    public static bool ShouldDrop(float dropChance, double roll)
    {
        if (dropChance <= 0f)
            return false;
        if (dropChance >= 1f)
            return true;
        return roll < dropChance;
    }
}
```

```csharp
public sealed class DroppedItem2D : MonoBehaviour
{
    [SerializeField] private SpriteRenderer visual;
    [SerializeField] private CircleCollider2D pickupZone;

    public ItemDefinition Item { get; private set; }

    /// 渲染贴图取 worldSprite，缺失时回退 icon；体量按「最大边」等比缩放到 item.WorldSize。
    /// 只调整体量，不改变贴图比例 —— 细长的剑保持细长，竖立的箱盒保持竖立。
    public void Initialize(ItemDefinition item)
    {
        Item = item;

        if (item == null)
        {
            DisableVisual();
            return;
        }

        var sprite = item.WorldSprite != null ? item.WorldSprite : item.Icon;
        if (visual == null || sprite == null)
        {
            DisableVisual();
            return;
        }

        visual.sprite = sprite;
        visual.enabled = true;

        var bounds = sprite.bounds.size;
        var maxSide = Mathf.Max(bounds.x, bounds.y);
        if (maxSide > 0f && item.WorldSize > 0f)
            visual.transform.localScale = Vector3.one * (item.WorldSize / maxSide);

        if (pickupZone != null)
            pickupZone.radius = item.WorldSize * 0.5f;
    }

    private void DisableVisual()
    {
        if (visual != null)
            visual.enabled = false;
    }
```

```csharp
using System.Collections.Generic;
using UnityEngine;

public static class LootScan
{
    /// 半径内是否存在存活敌人（已死亡的敌人不算）。
    public static bool HasAliveEnemyWithin(EnemyBrain2D[] enemies, Vector2 origin, float radius)
    {
        if (enemies == null)
            return false;

        var sqrRadius = radius * radius;
        foreach (var enemy in enemies)
        {
            if (enemy == null || enemy.IsDead)
                continue;

            var delta = (Vector2)enemy.transform.position - origin;
            if (delta.sqrMagnitude < sqrRadius)
                return true;
        }

        return false;
    }

    /// 按到 origin 的距离升序排序，截断到 max 个。
    public static IReadOnlyList<DroppedItem2D> SelectNearest(DroppedItem2D[] candidates, Vector2 origin, int max)
    {
        var result = new List<DroppedItem2D>();
        if (candidates == null || max <= 0)
            return result;

        var sorted = new List<DroppedItem2D>();
        foreach (var candidate in candidates)
        {
            if (candidate != null)
                sorted.Add(candidate);
        }

        sorted.Sort((a, b) =>
            ((Vector2)a.transform.position - origin).sqrMagnitude
                .CompareTo(((Vector2)b.transform.position - origin).sqrMagnitude));

        for (var i = 0; i < sorted.Count && i < max; i++)
            result.Add(sorted[i]);

        return result;
    }
}
```

```csharp
using UnityEngine;

public sealed class LootDropper : MonoBehaviour
{
    [SerializeField] private EnemyBrain2D enemy;
    [SerializeField] private LootTable lootTable;
    [SerializeField] private DroppedItem2D dropPrefab;
    [SerializeField] private float scatterRadius = 0.35f;

    private bool subscribed;

    private void Awake()
    {
        enemy ??= GetComponent<EnemyBrain2D>();
        Subscribe();
    }

    private void OnDestroy()
    {
        if (!subscribed || enemy == null)
            return;

        enemy.Died -= HandleDied;
        subscribed = false;
    }

    /// 幂等订阅；测试可直接调用，避免依赖 EditMode 中不会触发的 Awake。
    public void Subscribe()
    {
        if (subscribed)
            return;

        if (enemy == null)
            enemy = GetComponent<EnemyBrain2D>();
        if (enemy == null)
            return;

        enemy.Died += HandleDied;
        subscribed = true;
    }

    /// 敌人死亡时由 Died 事件调用；可直接调用以便测试。
    public void HandleDied()
    {
        if (lootTable == null || dropPrefab == null)
            return;

        var items = lootTable.Roll(new System.Random());
        foreach (var item in items)
        {
            var offset = Random.insideUnitCircle * scatterRadius;
            var position = (Vector2)transform.position + offset;
            var dropped = Instantiate(dropPrefab, position, Quaternion.identity);
            dropped.Initialize(item);
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using TMPro;
using UnityEngine;

public sealed class LootPanelController : MonoBehaviour
{
    [SerializeField] private GameObject panelRoot;
    [SerializeField] private LootRowView[] rows;
    [SerializeField] private TMP_Text hintText;
    [SerializeField] private float hintDuration = 1.5f;

    private float hintRemaining;

    public event Action<int> ItemSelected;

    public bool IsVisible => panelRoot != null && panelRoot.activeSelf;

    private void Awake()
    {
        Hide();
    }

    private void Update()
    {
        if (hintRemaining <= 0f)
            return;

        hintRemaining = Mathf.Max(0f, hintRemaining - Time.unscaledDeltaTime);
        if (hintRemaining <= 0f && hintText != null)
            hintText.text = string.Empty;
    }

    public void Show(IReadOnlyList<LootEntry> entries)
    {
        if (panelRoot == null)
            return;

        BindRows();
        panelRoot.SetActive(true);

        var count = entries != null ? entries.Count : 0;
        for (var i = 0; i < rows.Length; i++)
        {
            if (rows[i] == null)
                continue;

            if (i < count)
                rows[i].SetEntry(entries[i], true);
            else
                rows[i].SetEntry(default, false);
        }
    }

    public void Hide()
    {
        if (panelRoot != null)
            panelRoot.SetActive(false);
    }

    public void ShowHint(string message)
    {
        if (hintText == null)
            return;

        hintText.text = message;
        hintRemaining = Mathf.Max(0f, hintDuration);
    }

    private void BindRows()
    {
        if (rows == null)
            return;

        for (var i = 0; i < rows.Length; i++)
        {
            if (rows[i] != null)
                rows[i].Bind(i, RaiseSelected);
        }
    }

    private void RaiseSelected(int index)
    {
        ItemSelected?.Invoke(index);
    }
}
```

```csharp
using System.Collections.Generic;
using UnityEngine;

public sealed class LootInteractor : MonoBehaviour
{
    [SerializeField] private float pickupRadius = 1.6f;
    [SerializeField] private float combatRadius = 8f;
    [SerializeField] private int maxEntries = 4;
    [SerializeField] private float scanInterval = 0.2f;
    [SerializeField] private InventoryHolder inventory;
    [SerializeField] private LootPanelController panel;

    private readonly List<DroppedItem2D> current = new List<DroppedItem2D>();
    private readonly List<DroppedItem2D> previous = new List<DroppedItem2D>();
    private readonly List<LootEntry> entries = new List<LootEntry>();
    private float timer;
    private bool panelVisible;

    private void Awake()
    {
        if (panel != null)
            panel.ItemSelected += HandleItemSelected;
    }

    private void OnDestroy()
    {
        if (panel != null)
            panel.ItemSelected -= HandleItemSelected;
    }

    private void Update()
    {
        timer -= Time.deltaTime;
        if (timer > 0f)
            return;

        timer = Mathf.Max(0.01f, scanInterval);
        Scan();
    }

    /// 一轮扫描：先判附近有没有存活敌人，没有才收集掉落物并刷新面板。
    public void Scan()
    {
        if (panel == null)
            return;

        var enemies = FindObjectsOfType<EnemyBrain2D>();
        if (LootScan.HasAliveEnemyWithin(enemies, transform.position, combatRadius))
        {
            HidePanel();
            return;
        }

        current.Clear();
        var hits = Physics2D.OverlapCircleAll(transform.position, pickupRadius);
        foreach (var hit in hits)
        {
            var dropped = hit.GetComponentInParent<DroppedItem2D>();
            if (dropped == null || current.Contains(dropped))
                continue;

            current.Add(dropped);
        }

        var nearest = LootScan.SelectNearest(current.ToArray(), transform.position, maxEntries);
        current.Clear();
        current.AddRange(nearest);

        if (current.Count == 0)
        {
            HidePanel();
            return;
        }

        if (SameAsPrevious())
        {
            panelVisible = true;
            return;
        }

        BuildEntries();
        previous.Clear();
        previous.AddRange(current);
        panelVisible = true;
        panel.Show(entries);
    }

    /// 拾取的唯一入口。将来接联机时只改这个方法内部：发请求 + 服务器校验后再发货。
    public bool TryPickup(DroppedItem2D dropped)
    {
        if (dropped == null || inventory == null || dropped.Item == null)
            return false;

        if (!inventory.Model.TryAdd(dropped.Item))
        {
            if (panel != null)
                panel.ShowHint("背包已满");

            return false;
        }

        Destroy(dropped.gameObject);
        previous.Clear();
        Scan();
        return true;
    }

    private void HandleItemSelected(int index)
    {
        if (index < 0 || index >= current.Count)
            return;

        TryPickup(current[index]);
    }

    private bool SameAsPrevious()
    {
        if (previous.Count != current.Count)
            return false;

        for (var i = 0; i < current.Count; i++)
        {
            if (previous[i] != current[i])
                return false;
        }

        return true;
    }

    private void BuildEntries()
    {
        entries.Clear();
        foreach (var dropped in current)
        {
            var item = dropped.Item;
            entries.Add(new LootEntry
            {
                DisplayName = item != null ? item.DisplayName : string.Empty,
                Value = item != null ? item.Value : 0,
                Icon = item != null ? item.Icon : null,
            });
        }
    }

    private void HidePanel()
    {
        previous.Clear();
        if (!panelVisible)
            return;

        panelVisible = false;
        panel.Hide();
    }
}
```
